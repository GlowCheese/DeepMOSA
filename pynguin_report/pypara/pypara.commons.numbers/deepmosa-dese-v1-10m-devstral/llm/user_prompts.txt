def sign(x: Numeric) -> int:
    """
    Defines a sign predicate for numeric values (ints, floats and Decimals).

    :param number: A number.
    :return: The sign of the given number.

    >>> sign(1)
    1
    >>> sign(0)
    0
    >>> sign(-0)
    0
    >>> sign(-1)
    -1
    >>> sign(Decimal("1"))
    1
    >>> sign(Decimal("0"))
    0
    >>> sign(-Decimal("0"))
    0
    >>> sign(Decimal("-1"))
    -1
    """
